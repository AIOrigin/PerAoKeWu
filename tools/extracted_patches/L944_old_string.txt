func _update_train_blade_gates() -> void:
	var near_lo := track_distance - 8.0
	var near_hi := track_distance + 70.0
	for obstacle in obstacles:
		var otype := String(obstacle.get("type", ""))
		if otype not in ["train", "train_moving"]:
			continue
		var dist := float(obstacle.get("distance", 0.0)) + float(obstacle.get("move_offset", 0.0))
		if dist < near_lo or dist > near_hi:
			continue
		var node := obstacle.get("node") as Node3D
		if node == null or not is_instance_valid(node):
			continue
		obstacle["blade_top"] = float(node.get_meta("blade_top", TRAIN_GATE_TOP))
		var blade := node.get_node_or_null("TrainWaveBlade") as MeshInstance3D
		if blade == null:
			blade = _add_train_wave_blade(node, TRAIN_GATE_BLADE_SPAN)
		var top_y := float(obstacle.get("blade_top", TRAIN_GATE_TOP))
		var y := _slide_blade_local_y(obstacle)
		blade.position.y = y
		var blocking := _slide_blade_is_blocking(obstacle)
		var flicker := 0.75 + 0.25 * absf(sin(Time.get_ticks_msec() * 0.045 + dist * 0.3))
		var crackle := 0.85 + 0.15 * absf(sin(Time.get_ticks_msec() * 0.11 + dist))
		blade.scale = Vector3(1.0, (1.55 if blocking else 0.95) * crackle, 1.0 + 0.08 * flicker)
		var mat := blade.get_active_material(0) as StandardMaterial3D
		if mat:
			# 落下挡路：血红刀；上方可过：金黄光刃；持续闪烁电弧感
			mat.albedo_color = Color(1.0, 0.28, 0.32, 0.96) if blocking else Color(1.0, 0.88, 0.4, 0.95)
			mat.emission = Color(1.0, 0.12, 0.18) if blocking else Color(1.0, 0.78, 0.22)
			mat.emission_energy_multiplier = (9.5 if blocking else 7.2) * flicker
		var trail := node.get_node_or_null("TrainWaveBladeTrail") as MeshInstance3D
		if trail:
			trail.visible = blocking or _slide_blade_phase(obstacle) < 0.42
			trail.position.y = lerpf(y, top_y - 0.2, 0.42)
			var th := maxf(top_y - y, 0.4)
			trail.scale = Vector3(1.0, th / 1.6, 1.0)
			var trail_mat := trail.get_active_material(0) as StandardMaterial3D
			if trail_mat:
				trail_mat.emission_energy_multiplier = (5.5 if blocking else 3.2) * flicker
		var arc := node.get_node_or_null("TrainWaveBladeArc") as MeshInstance3D
		if arc:
			arc.position.y = y
			arc.visible = true
			arc.scale = Vector3(1.0 + 0.12 * flicker, 0.7 + 0.9 * flicker, 1.0 + 0.35 * crackle)
			var arc_mat := arc.get_active_material(0) as StandardMaterial3D
			if arc_mat:
				arc_mat.albedo_color = Color(1.0, 0.55, 0.35, 0.42 + 0.35 * flicker) if blocking \
					else Color(1.0, 0.95, 0.55, 0.35 + 0.4 * flicker)
				arc_mat.emission = Color(1.0, 0.25, 0.15) if blocking else Color(1.0, 0.92, 0.4)
				arc_mat.emission_energy_multiplier = (12.0 if blocking else 9.0) * flicker
		var sparks := node.get_node_or_null("TrainWaveBladeSparks") as GPUParticles3D
		if sparks:
			sparks.position.y = y
			sparks.emitting = true
			sparks.amount_ratio = 1.0 if blocking else 0.55
			sparks.speed_scale = 1.6 if blocking else 1.15
